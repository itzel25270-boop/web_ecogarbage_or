#include <WiFi.h>
#include <ESPAsyncWebServer.h>
#include <ArduinoJson.h>

// Credenciales de red Wi-Fi
const char* ssid = "TU_RED_WIFI";
const char* password = "TU_CONTRASEÑA";

// Pines de Hardware ESP32 para Ecogarbage
const int PIN_BOMBA = 26;          // Relay / Transistor Mosfet para Bomba
const int PIN_SENSOR_HUMEDAD = 34; // Pin ADC Analógico Humedad del Suelo
const int PIN_BATERIA = 35;        // Pin ADC Analógico Divisor de Voltaje

AsyncWebServer server(80);
AsyncWebSocket ws("/ws");

bool bombaActiva = false;
bool modoAutomatico = true;
int umbralHumedad = 35;
unsigned long ultimoEnvio = 0;

String obtenerEstadoJSON() {
  // Lectura del sensor capacitivo de humedad
  int rawHumedad = analogRead(PIN_SENSOR_HUMEDAD);
  int humedadPct = map(rawHumedad, 4095, 1500, 0, 100); 
  humedadPct = constrain(humedadPct, 0, 100);

  // Lectura del voltaje y porcentaje de batería Li-Ion
  int rawBat = analogRead(PIN_BATERIA);
  float voltajeBat = (rawBat / 4095.0) * 2.0 * 3.3; // Factor de divisor 1:1
  int bateriaPct = map((int)(voltajeBat * 100), 330, 420, 0, 100);
  bateriaPct = constrain(bateriaPct, 0, 100);

  // Lógica de automatización local en ESP32
  if (modoAutomatico) {
    if (humedadPct < umbralHumedad && !bombaActiva) {
      bombaActiva = true;
      digitalWrite(PIN_BOMBA, HIGH);
    } else if (humedadPct >= umbralHumedad + 10 && bombaActiva) {
      bombaActiva = false;
      digitalWrite(PIN_BOMBA, LOW);
    }
  }

  StaticJsonDocument<250> doc;
  doc["humedad"] = humedadPct;
  doc["bateriaPct"] = bateriaPct;
  doc["bateriaVolt"] = String(voltajeBat, 2);
  doc["bomba"] = bombaActiva;
  doc["auto"] = modoAutomatico;
  doc["umbral"] = umbralHumedad;

  String output;
  serializeJson(doc, output);
  return output;
}

void procesarMensajeWS(void *arg, uint8_t *data, size_t len) {
  AwsFrameInfo *info = (AwsFrameInfo*)arg;
  if (info->final && info->index == 0 && info->len == len && info->opcode == WS_TEXT) {
    data[len] = 0;
    StaticJsonDocument<200> doc;
    DeserializationError error = deserializeJson(doc, (char*)data);
    if (!error) {
      if (doc.containsKey("bomba")) {
        bombaActiva = doc["bomba"];
        digitalWrite(PIN_BOMBA, bombaActiva ? HIGH : LOW);
      }
      if (doc.containsKey("auto")) modoAutomatico = doc["auto"];
      if (doc.containsKey("umbral")) umbralHumedad = doc["umbral"];
      ws.textAll(obtenerEstadoJSON());
    }
  }
}

void onEvent(AsyncWebSocket *server, AsyncWebSocketClient *client, AwsEventType type, void *arg, uint8_t *data, size_t len) {
  if (type == WS_EVT_CONNECT) {
    client->text(obtenerEstadoJSON());
  } else if (type == WS_EVT_DATA) {
    procesarMensajeWS(arg, data, len);
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_BOMBA, OUTPUT);
  digitalWrite(PIN_BOMBA, LOW);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\n[Ecogarbage] WiFi Conectado! IP: " + WiFi.localIP().toString());

  ws.onEvent(onEvent);
  server.addHandler(&ws);
  server.begin();
}

void loop() {
  ws.cleanupClients();
  // Transmitir telemetría cada 2 segundos a clientes conectados
  if (millis() - ultimoEnvio > 2000) {
    ws.textAll(obtenerEstadoJSON());
    ultimoEnvio = millis();
  }
}
